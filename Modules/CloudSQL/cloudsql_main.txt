# PostgreSQL database with environment-specific configurations
# =============================================================================

# Random password generation for database users
resource "random_password" "postgres_password" {
  length  = 16
  special = true
  # Use characters that are safe for database passwords
  override_special = "!#$%&*()-_=+[]{}<>?"
  
  keepers = {
    instance_name = var.instance_name
    environment   = var.environment
  }
}

resource "random_password" "app_password" {
  length  = 16
  special = true
  override_special = "!#$%&*()-_=+[]{}<>?"
  
  keepers = {
    instance_name = var.instance_name
    environment   = var.environment
    app_user      = var.app_user_name
  }
}

# =============================================================================
# CLOUDSQL INSTANCE
# =============================================================================

resource "google_sql_database_instance" "postgres" {
  name                = var.instance_name
  database_version    = var.database_version
  region             = var.region
  deletion_protection = var.deletion_protection

  # Depend on private networking setup
  depends_on = [
    google_service_networking_connection.private_vpc_connection
  ]

  settings {
    # Tier configuration (environment-specific)
    tier = var.tier

    # Disk configuration
    disk_size       = var.disk_size_gb
    disk_type       = var.disk_type
    disk_autoresize = var.disk_autoresize
    
    # Autoresize limit (prevent runaway costs)
    disk_autoresize_limit = var.disk_autoresize_limit

    # Availability type (ZONAL for dev, REGIONAL for prod)
    availability_type = var.availability_type

    # IP configuration (private networking)
    ip_configuration {
      ipv4_enabled                                  = var.enable_public_ip
      private_network                              = var.network_self_link
      enable_private_path_for_google_cloud_services = true
      require_ssl                                  = var.require_ssl

      # Authorized networks (only if public IP is enabled)
      dynamic "authorized_networks" {
        for_each = var.enable_public_ip ? var.authorized_networks : []
        content {
          name  = authorized_networks.value.name
          value = authorized_networks.value.value
        }
      }
    }

    # Backup configuration (environment-specific)
    backup_configuration {
      enabled                        = var.backup_enabled
      start_time                     = var.backup_start_time
      location                       = var.backup_location
      point_in_time_recovery_enabled = var.point_in_time_recovery_enabled
      transaction_log_retention_days = var.transaction_log_retention_days
      
      # Backup retention settings
      backup_retention_settings {
        retained_backups = var.backup_retained_backups
        retention_unit   = "COUNT"
      }
    }

    # Maintenance window (prod only)
    dynamic "maintenance_window" {
      for_each = var.environment == "prod" ? [1] : []
      content {
        day          = var.maintenance_window_day
        hour         = var.maintenance_window_hour
        update_track = var.maintenance_window_update_track
      }
    }

    # Database flags for performance tuning
    dynamic "database_flags" {
      for_each = var.database_flags
      content {
        name  = database_flags.value.name
        value = database_flags.value.value
      }
    }

    # Insights configuration (monitoring)
    insights_config {
      query_insights_enabled  = var.query_insights_enabled
      query_string_length     = var.query_string_length
      record_application_tags = var.record_application_tags
      record_client_address   = var.record_client_address
    }

    # Connector enforcement (for secure connections)
    connector_enforcement = var.connector_enforcement

    # Location preference
    location_preference {
      zone                   = var.preferred_zone
      secondary_zone         = var.secondary_zone
      follow_gae_application = var.follow_gae_application
    }

    # User labels for cost tracking
    user_labels = merge(
      var.labels,
      {
        environment    = var.environment
        instance-name  = var.instance_name
        managed-by     = "terraform"
        database-type  = "postgresql"
        cost-center    = var.environment == "prod" ? "production" : "development"
      }
    )
  }

  # Timeouts for long-running operations
  timeouts {
    create = "20m"
    update = "20m"
    delete = "20m"
  }

  # Prevent accidental deletion in production
  lifecycle {
    prevent_destroy = var.environment == "prod" ? true : false
  }
}

# =============================================================================

# Main application database
resource "google_sql_database" "main_database" {
  name      = var.database_name
  instance  = google_sql_database_instance.postgres.name
  charset   = var.database_charset
  collation = var.database_collation

  # Prevent accidental deletion
  lifecycle {
    prevent_destroy = var.environment == "prod" ? true : false
  }
}

# Additional databases (if specified)
resource "google_sql_database" "additional_databases" {
  for_each = toset(var.additional_databases)

  name      = each.value
  instance  = google_sql_database_instance.postgres.name
  charset   = var.database_charset
  collation = var.database_collation
}

# =============================================================================
# DATABASE USERS
# =============================================================================

# Default postgres superuser
resource "google_sql_user" "postgres_user" {
  name     = "postgres"
  instance = google_sql_database_instance.postgres.name
  password = random_password.postgres_password.result
  type     = "BUILT_IN"

  # For PostgreSQL, we can set host to restrict connections
  host = var.restrict_postgres_user ? "%" : null
}

# Application user with limited privileges
resource "google_sql_user" "app_user" {
  name     = var.app_user_name
  instance = google_sql_database_instance.postgres.name
  password = random_password.app_password.result
  type     = "BUILT_IN"
}

# Additional users (if specified)
resource "google_sql_user" "additional_users" {
  for_each = var.additional_users

  name     = each.key
  instance = google_sql_database_instance.postgres.name
  password = each.value.password != null ? each.value.password : random_password.app_password.result
  type     = each.value.type
  host     = each.value.host
}

# =============================================================================
# SSL CERTIFICATES (if SSL is required)
# =============================================================================

resource "google_sql_ssl_cert" "client_cert" {
  count = var.require_ssl && var.create_ssl_cert ? 1 : 0

  common_name = "${var.instance_name}-client-cert"
  instance    = google_sql_database_instance.postgres.name

  depends_on = [google_sql_database_instance.postgres]
}

# =============================================================================
# DATABASE REPLICA (for production high availability)
# =============================================================================

resource "google_sql_database_instance" "read_replica" {
  count = var.environment == "prod" && var.enable_read_replica ? 1 : 0

  name                 = "${var.instance_name}-replica"
  database_version     = var.database_version
  region              = var.replica_region != null ? var.replica_region : var.region
  master_instance_name = google_sql_database_instance.postgres.name

  replica_configuration {
    failover_target = var.replica_failover_target
  }

  settings {
    tier              = var.replica_tier != null ? var.replica_tier : var.tier
    availability_type = "ZONAL" # Replicas are always zonal
    disk_autoresize   = var.disk_autoresize

    ip_configuration {
      ipv4_enabled                                  = false
      private_network                              = var.network_self_link
      enable_private_path_for_google_cloud_services = true
    }

    # Read replicas have limited configuration options
    backup_configuration {
      enabled = false # Replicas don't need their own backups
    }

    user_labels = merge(
      var.labels,
      {
        environment   = var.environment
        instance-type = "read-replica"
        master        = var.instance_name
        managed-by    = "terraform"
      }
    )
  }

  depends_on = [google_sql_database_instance.postgres]
}
